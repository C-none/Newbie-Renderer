file(GLOB_RECURSE NR_SHADER_SLANG_FILES CONFIGURE_DEPENDS "*.slang")

add_custom_target(ShaderAssets SOURCES ${NR_SHADER_SLANG_FILES})

    FOLDER "shader"
)
