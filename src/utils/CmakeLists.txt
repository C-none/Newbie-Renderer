# ============================================================================
# Build utils library
# ============================================================================

if(NOT NR_LOG_LEVEL MATCHES "^(INFO|WARNING|ERROR)$")
    message(FATAL_ERROR "NR_LOG_LEVEL must be INFO, WARNING, or ERROR.")
endif()

set(NR_STATIC_UTILS_CONSTANTS_CONFIG "${CMAKE_CURRENT_BINARY_DIR}/generated/$<CONFIG>/staticUtilsConstantsConfig.inl")

add_custom_command(
    OUTPUT "${NR_STATIC_UTILS_CONSTANTS_CONFIG}"
    COMMAND ${CMAKE_COMMAND}
        -D "NR_CACHE_FILE=${CMAKE_BINARY_DIR}/CMakeCache.txt"
        -D "NR_SOURCE_DIR=${CMAKE_SOURCE_DIR}"
        -D "NR_BINARY_DIR=${CMAKE_BINARY_DIR}"
        -D "NR_BUILD_CONFIG=$<CONFIG>"
        -D "NR_OUTPUT_FILE=${NR_STATIC_UTILS_CONSTANTS_CONFIG}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/GenerateStaticUtilsConstantsConfig.cmake"
    DEPENDS
        "${CMAKE_BINARY_DIR}/CMakeCache.txt"
        "${CMAKE_CURRENT_SOURCE_DIR}/GenerateStaticUtilsConstantsConfig.cmake"
    COMMENT "Generating staticUtilsConstantsConfig.inl (BUILD_CONFIG=$<CONFIG>)"
    VERBATIM
)

add_custom_target(generate_static_utils_constants_config
    DEPENDS "${NR_STATIC_UTILS_CONSTANTS_CONFIG}"
)

file(GLOB MODULE_UNITS
    CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/*.ixx"
)

nr_add_library(utils STATIC)
add_dependencies(utils generate_static_utils_constants_config)
target_link_libraries(utils PUBLIC
    dependency
)
target_include_directories(utils PRIVATE
    "${CMAKE_CURRENT_BINARY_DIR}/generated/$<CONFIG>"
)
target_sources(utils
    PUBLIC
        FILE_SET cxx_modules TYPE CXX_MODULES FILES ${MODULE_UNITS}
)

source_group(TREE ${CMAKE_CURRENT_SOURCE_DIR} FILES
    ${MODULE_UNITS}
    # src/${IMPL_SOURCES}
)
