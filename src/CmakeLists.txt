add_subdirectory(extern)

# utils must be before any other subdirectory that might depend on it
add_subdirectory(utils)
add_subdirectory(load)
add_subdirectory(resource)
add_subdirectory(scene)

add_subdirectory(rhi)
add_subdirectory(renderer)
add_subdirectory(renderPasses)

file(GLOB IMPL_SOURCES
    CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp"
)

nr_add_executable(main main.cpp)
target_link_libraries(main PRIVATE 
    nrrenderer
    nrrenderpasses
    dependency
)
target_sources(main
	PRIVATE
		${IMPL_SOURCES}
)

source_group(TREE ${CMAKE_CURRENT_SOURCE_DIR} FILES
    ${IMPL_SOURCES}
)

add_custom_target(run_main
    COMMAND $<TARGET_FILE:main>
    DEPENDS main
    WORKING_DIRECTORY $<TARGET_FILE_DIR:main>
    COMMENT "Build and run main executable"
    USES_TERMINAL
)
