function(nr_apply_msvc_settings target)
    # WINDOWS-FLAG
    if(MSVC)
        target_compile_options(${target} PRIVATE /MP)
        set_target_properties(${target} PROPERTIES INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
        target_compile_options(${target} PRIVATE
            $<$<CONFIG:Debug>:/W4>
            $<$<CONFIG:Debug>:/WX>
            $<$<CONFIG:Release>:/W4>
            $<$<CONFIG:Release>:/WX>
            # $<$<CONFIG:Debug>:/fsanitize=address> # Only enable when encountering memory issues(e.g. Segment fault). At the same time, disable INCREMENTAL flag if this flag is used.
        )
        target_link_options(${target} PRIVATE
            $<$<CONFIG:Debug>:/INCREMENTAL>
            $<$<CONFIG:Debug>:/DEBUG:FULL>
            $<$<CONFIG:Release>:/INCREMENTAL:NO>
            $<$<CONFIG:RelWithDebInfo>:/INCREMENTAL:NO>)
    endif()
endfunction()


function(nr_add_executable target)
    add_executable(${target} ${ARGN})
    nr_apply_msvc_settings(${target})
endfunction()

function(nr_add_library target)
    add_library(${target} ${ARGN})
    nr_apply_msvc_settings(${target})
endfunction()
