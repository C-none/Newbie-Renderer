function(nr_apply_msvc_settings target)
    # WINDOWS-FLAG
    if(MSVC)
        # Modern parallel compilation
        target_compile_options(${target} PRIVATE /MP)
        
        # Link-time optimization for Release builds
        set_target_properties(${target} PROPERTIES
            INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE
            INTERPROCEDURAL_OPTIMIZATION_RELWITHDEBINFO TRUE
        )
        
        # Warning levels using generator expressions (cleaner syntax)
        target_compile_options(${target} PRIVATE
            $<$<CONFIG:Debug>:/W4 /WX /Zi>           # All warnings as errors, debug info
            $<$<CONFIG:Release>:/W4 /WX /Ox>          # All warnings as errors, optimize
            $<$<CONFIG:RelWithDebInfo>:/W4 /WX /Zi>  # Warnings + debug info
            # $<$<CONFIG:Debug>:/fsanitize=address> # Only enable when encountering memory issues(e.g. Segment fault). At the same time, disable INCREMENTAL flag if this flag is used.
        )
        
        # Incremental linking for faster rebuilds
        target_link_options(${target} PRIVATE
            $<$<CONFIG:Debug>:/INCREMENTAL /DEBUG:FULL>
            $<$<CONFIG:Release>:/INCREMENTAL:NO /OPT:REF /OPT:ICF>
            $<$<CONFIG:RelWithDebInfo>:/INCREMENTAL:NO /DEBUG:FULL>
        )
        
        # Runtime library selection
        # set_target_properties(${target} PROPERTIES MSVC_RUNTIME_LIBRARY 
        #     MultiThreaded$<$<CONFIG:Debug>:Debug>)
    endif()
endfunction()

function(nr_add_executable target)
    add_executable(${target} ${ARGN})
    nr_apply_msvc_settings(${target})
endfunction()

function(nr_add_library target)
    add_library(${target} ${ARGN})
    nr_apply_msvc_settings(${target})
endfunction()
