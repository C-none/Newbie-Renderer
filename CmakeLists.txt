function(nr_apply_target_settings target)
    if(MSVC)
        # Modern parallel compilation
        target_compile_options(${target} PRIVATE /MP /EHsc)
        target_compile_definitions(${target} PRIVATE
            $<$<CONFIG:Debug>:_ITERATOR_DEBUG_LEVEL=2>
            $<$<NOT:$<CONFIG:Debug>>:_ITERATOR_DEBUG_LEVEL=0>
        )

        if(NR_RELEASE_IPO_SUPPORTED)
            set_target_properties(${target} PROPERTIES
                INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE
            )
        endif()

        target_compile_options(${target} PRIVATE
            $<$<CONFIG:Debug>:/W4>
            $<$<CONFIG:Debug>:/WX>
            $<$<CONFIG:Debug>:/Zi>
            $<$<CONFIG:Release>:/W4>
            $<$<CONFIG:Release>:/WX>
            $<$<CONFIG:Release>:/O2>
        )

        target_link_options(${target} PRIVATE
            $<$<CONFIG:Debug>:/INCREMENTAL>
            $<$<CONFIG:Debug>:/DEBUG:FULL>
            $<$<CONFIG:Release>:/INCREMENTAL:NO>
            $<$<CONFIG:Release>:/OPT:REF>
            $<$<CONFIG:Release>:/OPT:ICF>
        )
    elseif(CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "GNU")
        if(NR_RELEASE_IPO_SUPPORTED)
            set_target_properties(${target} PROPERTIES
                INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE
            )
        endif()

        target_compile_options(${target} PRIVATE
            -Wall
            -Wextra
            -Werror
            -Wno-error=reserved-module-identifier
            $<$<CONFIG:Debug>:-Og>
            $<$<CONFIG:Debug>:-g3>
            $<$<CONFIG:Debug>:-fno-omit-frame-pointer>
            $<$<CONFIG:Release>:-O3>
            $<$<CONFIG:Release>:-ffunction-sections>
            $<$<CONFIG:Release>:-fdata-sections>
        )

        target_link_options(${target} PRIVATE
            $<$<CONFIG:Release>:-fuse-ld=lld>
            $<$<CONFIG:Release>:-Wl,--gc-sections>
            $<$<CONFIG:Release>:-Wl,--icf=all>
        )
    endif()
endfunction()

function(nr_add_slang_runtime_dependency target)
    set(options DEFER_TARGET_DEPENDENCY)
    set(one_value_args OUT_COPY_STAMP)
    cmake_parse_arguments(PARSE_ARGV 1 arg "${options}" "${one_value_args}" "")

    if(NOT WIN32 OR NOT DEFINED NR_SLANG_RUNTIME_DIR)
        if(arg_OUT_COPY_STAMP)
            set(${arg_OUT_COPY_STAMP} "" PARENT_SCOPE)
        endif()
        return()
    endif()

    set(runtime_output_dir "${CMAKE_CURRENT_BINARY_DIR}")
    if(CMAKE_CONFIGURATION_TYPES)
        string(APPEND runtime_output_dir "/$<CONFIG>")
    endif()

    string(SHA256 runtime_copy_hash "${CMAKE_CURRENT_BINARY_DIR}")
    string(SUBSTRING "${runtime_copy_hash}" 0 16 runtime_copy_hash)
    set(runtime_copy_target "nr_copy_slang_runtime_${runtime_copy_hash}")
    set(runtime_copy_stamp "${CMAKE_CURRENT_BINARY_DIR}/CMakeFiles/${runtime_copy_target}_$<CONFIG>.stamp")

    if(NOT TARGET ${runtime_copy_target})
        set(runtime_copy_dependencies)
        if(DEFINED NR_SLANG_RUNTIME_STAMP)
            list(APPEND runtime_copy_dependencies "${NR_SLANG_RUNTIME_STAMP}")
        endif()

        add_custom_command(
            OUTPUT "${runtime_copy_stamp}"
            COMMAND ${CMAKE_COMMAND} -E copy_directory
                "${NR_SLANG_RUNTIME_DIR}"
                "${runtime_output_dir}"
            COMMAND ${CMAKE_COMMAND} -E touch "${runtime_copy_stamp}"
            DEPENDS ${runtime_copy_dependencies}
            COMMENT "Copying Slang runtime files to ${runtime_output_dir}"
            VERBATIM
        )
        add_custom_target(${runtime_copy_target}
            DEPENDS "${runtime_copy_stamp}"
        )
    endif()

    if(arg_OUT_COPY_STAMP)
        set(${arg_OUT_COPY_STAMP} "${runtime_copy_stamp}" PARENT_SCOPE)
    endif()

    if(NOT arg_DEFER_TARGET_DEPENDENCY)
        add_dependencies(${target} ${runtime_copy_target})
    endif()
endfunction()

function(nr_add_executable target)
    add_executable(${target} ${ARGN})
    if(NR_CMAKE_ENABLE_IMPORT_STD)
        set_target_properties(${target} PROPERTIES CXX_MODULE_STD ON)
    endif()
    nr_add_slang_runtime_dependency(${target})
    if(NR_DLSS_NGX_SDK_OPERATIONAL)
        if(TARGET nr_dlss_bridge)
            add_dependencies(${target} nr_dlss_bridge)
        endif()
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${NR_DLSS_BRIDGE_RUNTIME_DLL}"
                "$<TARGET_FILE_DIR:${target}>/nr_dlss_bridge.dll"
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${NR_DLSS_FEATURE_RUNTIME_DLL}"
                "$<TARGET_FILE_DIR:${target}>/nvngx_dlssd.dll"
            COMMENT "Copying the DLSS bridge and Ray Reconstruction runtime for ${target}"
            VERBATIM
        )
    endif()
    nr_apply_target_settings(${target})
endfunction()

function(nr_add_library target)
    add_library(${target} ${ARGN})
    if(NR_CMAKE_ENABLE_IMPORT_STD)
        set_target_properties(${target} PROPERTIES CXX_MODULE_STD ON)
    endif()
    nr_apply_target_settings(${target})
endfunction()

# GNU-style Clang single-config module builds can miss a source-level ordering
# edge for entry translation units that directly import modules produced by
# static library targets. target_link_libraries() records the module provider
# targets, but it does not always make the importing source object's compile
# step wait for the producer target's archive and module build work.
#
# Use this for a specific importing source file, typically an executable or test
# entry .cpp, when that file directly imports modules exported by static library
# targets. Do not use it as a generic replacement for target_link_libraries(),
# do not add it to every module interface, and do not add it for normal library
# module edges where CMake's CXX_MODULES file sets and dyndep scanning already
# model BMI ordering. It is unnecessary for MSVC, multi-config generators,
# non-static-library dependencies, and sources that do not import those modules.
function(nr_add_clang_module_object_depends source_file)
    if(NOT (CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "GNU"))
        return()
    endif()

    if(CMAKE_CONFIGURATION_TYPES)
        return()
    endif()

    set(module_object_depends "")
    foreach(dep_target IN LISTS ARGN)
        if(NOT TARGET ${dep_target})
            continue()
        endif()

        get_target_property(dep_target_type ${dep_target} TYPE)
        if(NOT dep_target_type STREQUAL "STATIC_LIBRARY")
            continue()
        endif()

        get_target_property(dep_target_binary_dir ${dep_target} BINARY_DIR)
        list(APPEND module_object_depends "${dep_target_binary_dir}/${CMAKE_STATIC_LIBRARY_PREFIX}${dep_target}${CMAKE_STATIC_LIBRARY_SUFFIX}")
    endforeach()

    if(module_object_depends)
        set_source_files_properties(${source_file} PROPERTIES OBJECT_DEPENDS "${module_object_depends}")
    endif()
endfunction()

# Convenience wrapper for executable/test entry sources that directly import
# project modules from dependency targets. Library targets should usually keep
# using target_link_libraries() and CXX_MODULES file sets unless they hit the
# specific GNU-style Clang single-config ordering issue described above.
function(nr_link_module_dependencies target source_file scope)
    set(allowed_scopes PRIVATE PUBLIC INTERFACE)
    if(NOT "${scope}" IN_LIST allowed_scopes)
        message(FATAL_ERROR "nr_link_module_dependencies scope must be PRIVATE, PUBLIC, or INTERFACE.")
    endif()

    if(NOT ARGN)
        message(FATAL_ERROR "nr_link_module_dependencies requires at least one dependency target.")
    endif()

    target_link_libraries(${target} ${scope} ${ARGN})
    nr_add_clang_module_object_depends(${source_file} ${ARGN})
endfunction()

function(nr_append_clangd_cache_arg args_var name value)
    string(REPLACE ";" "\\;" escaped_value "${value}")
    list(APPEND ${args_var} -D "${name}=${escaped_value}")
    set(${args_var} "${${args_var}}" PARENT_SCOPE)
endfunction()

if(NR_ENABLE_CLANGD_SYNC
   AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang"
   AND CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "GNU"
   AND CMAKE_GENERATOR MATCHES "^Ninja")
    set(nr_clangd_configure_args
        -S "${CMAKE_SOURCE_DIR}"
        -B "${NR_CLANGD_BINARY_DIR}"
        -G "${CMAKE_GENERATOR}"
        -D "CMAKE_CXX_STANDARD=${CMAKE_CXX_STANDARD}"
        -D "CMAKE_CXX_STANDARD_REQUIRED=${CMAKE_CXX_STANDARD_REQUIRED}"
        -D "CMAKE_CXX_EXTENSIONS=${CMAKE_CXX_EXTENSIONS}"
        -D "CMAKE_EXPORT_COMPILE_COMMANDS=ON"
        -D "NR_BUILD_TESTS=${NR_BUILD_TESTS}"
        -D "NR_ENABLE_RELEASE_IPO=${NR_ENABLE_RELEASE_IPO}"
        -D "NR_ENABLE_CLANGD_SYNC=OFF"
        -D "VCPKG_INSTALLED_DIR=${NR_CLANGD_BINARY_DIR}/vcpkg_installed"
    )

    if(CMAKE_TOOLCHAIN_FILE)
        nr_append_clangd_cache_arg(nr_clangd_configure_args CMAKE_TOOLCHAIN_FILE "${CMAKE_TOOLCHAIN_FILE}")
    endif()

    if(CMAKE_CONFIGURATION_TYPES)
        nr_append_clangd_cache_arg(nr_clangd_configure_args CMAKE_CONFIGURATION_TYPES "Debug")
        nr_append_clangd_cache_arg(nr_clangd_configure_args CMAKE_DEFAULT_BUILD_TYPE "Debug")
    else()
        nr_append_clangd_cache_arg(nr_clangd_configure_args CMAKE_BUILD_TYPE "Debug")
    endif()

    foreach(nr_clangd_cache_var IN ITEMS
        CMAKE_MAKE_PROGRAM
        CMAKE_C_COMPILER
        CMAKE_CXX_COMPILER
        CMAKE_CXX_COMPILER_CLANG_SCAN_DEPS
        CMAKE_C_COMPILER_TARGET
        CMAKE_CXX_COMPILER_TARGET
        CMAKE_AR
        CMAKE_RANLIB
        CMAKE_OBJCOPY
        CMAKE_STRIP
        CMAKE_RC_COMPILER
        VCPKG_FEATURE_FLAGS
        VCPKG_INSTALL_OPTIONS
        VCPKG_CHAINLOAD_TOOLCHAIN_FILE
        VCPKG_TARGET_TRIPLET
        VCPKG_HOST_TRIPLET
        VCPKG_TARGET_ARCHITECTURE
        VCPKG_OVERLAY_PORTS
        VCPKG_OVERLAY_TRIPLETS
    )
        if(DEFINED ${nr_clangd_cache_var})
            nr_append_clangd_cache_arg(nr_clangd_configure_args "${nr_clangd_cache_var}" "${${nr_clangd_cache_var}}")
        endif()
    endforeach()

    add_custom_target(nr_sync_clangd_database
        COMMAND ${CMAKE_COMMAND} ${nr_clangd_configure_args}
        COMMAND ${CMAKE_COMMAND} --build "${NR_CLANGD_BINARY_DIR}" --config Debug --target "${NR_CLANGD_SYNC_TARGET}"
        WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
        COMMENT "Synchronizing clangd compile database and Debug module BMI"
        USES_TERMINAL
        VERBATIM
    )
endif()
