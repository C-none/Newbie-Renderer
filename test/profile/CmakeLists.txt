function(nr_add_profile_contract_test target_name source_file)
    nr_add_executable(${target_name}
        ${source_file}
    )
    set_target_properties(${target_name} PROPERTIES FOLDER "Test/profile")
endfunction()

nr_add_profile_contract_test(nr_load_assimp_smoke_test nr_load_assimp_smoke_test.cpp)
target_link_libraries(nr_load_assimp_smoke_test PRIVATE
    nrload
    utils
    dependency
)
add_dependencies(nr_load_assimp_smoke_test nrload utils dependency)
nr_add_clang_module_object_depends(nr_load_assimp_smoke_test.cpp nrload utils dependency)
target_compile_definitions(nr_load_assimp_smoke_test PRIVATE
    NR_PROJECT_ROOT_DIR="${CMAKE_SOURCE_DIR}"
)
add_test(NAME nr_load_assimp_smoke_test COMMAND nr_load_assimp_smoke_test)
